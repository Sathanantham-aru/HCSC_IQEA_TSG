=== SECTION 0: BANNED PATTERNS (READ BEFORE GENERATING ANYTHING) ===
These are hard prohibitions. Violating any of these will produce a broken script.

❌ NEVER wrap a function definition inside `with allure.step(...):` at module level.
   allure.step as a context manager executes at import time when placed at module scope.
   This crashes the entire test suite before a single test runs.
   WRONG:
       with allure.step("creating helper"):
           def my_helper(driver): ...
   CORRECT:
       def my_helper(driver):
           with allure.step("using helper"): ...

❌ NEVER call switch_to_new_window() as the first step in a test before navigate().
   There is only one window handle at startup — this causes immediate RuntimeError.

❌ NEVER hardcode scope="session" on the driver fixture. Always use scope="function".

❌ NEVER pass a WebElement to EC.element_to_be_clickable() or EC.visibility_of_element_located().
   These only accept locator tuples: (By.XPATH, "...").
   The POM getter already handles waiting internally — call it directly, do not re-wrap.
   WRONG:  WebDriverWait(driver, 10).until(EC.element_to_be_clickable(element))
   CORRECT: element = pom.get_button()

❌ NEVER wrap a POM getter call inside another WebDriverWait.until() in the test file.
   POM getters already contain WebDriverWait(20) internally.
   A second outer WebDriverWait expires before the inner one completes → false TimeoutException.
   WRONG:  WebDriverWait(driver, 10).until(lambda d: pom.get_button())
   CORRECT: element = pom.get_button()

❌ NEVER pass a bare POM method reference to safe_send_keys or safe_click.
   Always wrap in a zero-argument lambda.
   WRONG:  safe_send_keys(driver, pom.get_username_input, "text")
   CORRECT: safe_send_keys(driver, lambda: pom.get_username_input(), "text")

❌ NEVER use dispatchEvent in the JS fallback inside safe_send_keys.
   Use ONLY: driver.execute_script("arguments[0].value = arguments[1];", fresh, text)
   Always re-fetch the element via getter_callable() before the JS call.

❌ NEVER invent dispatcher architectures like try_call_method, resolve_action,
   or dynamic method lookup via getattr/hasattr. Call POM methods directly by name.

❌ NEVER use a page-shell element (header, nav bar, cart icon) as the post-login
   readiness signal. Shell elements load before JS content hydrates.
   WRONG:  safe_get(lambda: inventory.get_shopping_cart_link())
   CORRECT: inventory.wait_until_page_ready()
            OR safe_get(lambda: inventory.get_add_to_cart_<first_product>())

❌ NEVER assert a URL keyword that described a previous step — assert the URL
   that is ACTUALLY true at the moment the assertion executes.
   WRONG after logout:    assert "inventory" in driver.current_url
   CORRECT after logout:  assert driver.current_url.rstrip("/") + "/" == BASE_URL
   WRONG after cart nav:  assert "inventory" in driver.current_url
   CORRECT after cart nav: assert "cart.html" in driver.current_url

def attach_screenshot(driver, name: str):
    try:
        png = driver.get_screenshot_as_png()
        allure.attach(png, name=name, attachment_type=AttachmentType.PNG)
        try:
            filepath = os.path.join(REPORTS_DIR, f"{{name.replace(' ','_')}}_{{int(time.time())}}.png")

                f.write(png)
        except Exception:
            pass
    except Exception:

HELPER 2 — safe_get (ALWAYS INCLUDE):
def safe_get(getter_callable, retries: int = 3, delay: float = 0.5):
    # Calls getter_callable() directly — NO outer WebDriverWait.
    # POM getter already contains WebDriverWait(20) internally.
    last_exc = None
    for _ in range(retries):
        try:
            elem = getter_callable()
            if elem is not None:
                return elem
        except Exception as e:
            last_exc = e
            time.sleep(delay)
    raise RuntimeError(f"safe_get failed after {{retries}} attempts: {{last_exc}}")

HELPER 3 — safe_click (ALWAYS INCLUDE):
def safe_click(driver, getter_callable, label: str = "", retries: int = 3):
    # getter_callable MUST be a zero-arg lambda: lambda: pom.get_button()
    # Re-fetches element before JS fallback — never reuses stale reference.
    last_exc = None
    for attempt in range(retries):
        try:
            elem = getter_callable()
            try:
                elem.click()
                return
            except (WebDriverException, StaleElementReferenceException) as e:
                last_exc = e
                fresh = getter_callable()
                driver.execute_script("arguments[0].click();", fresh)
                return
        except Exception as e:
            last_exc = e
            attach_screenshot(driver, f"safe_click_retry_{{label}}_{{attempt+1}}")
            time.sleep(0.5)
    attach_screenshot(driver, f"safe_click_failed_{{label}}")
    raise RuntimeError(f"safe_click [{{label}}] failed: {{last_exc}}")

HELPER 4 — safe_send_keys (ALWAYS INCLUDE):
def safe_send_keys(driver, getter_callable, text: str, label: str = "", retries: int = 3):
    # getter_callable MUST be a zero-arg lambda: lambda: pom.get_input()
    # JS fallback: .value assignment ONLY — dispatchEvent is banned.
    # Re-fetches element before JS fallback — never reuses stale reference.
    last_exc = None
    for attempt in range(retries):
        try:
            elem = getter_callable()
            try:
                elem.clear()
                elem.send_keys(text)
                return
            except (WebDriverException, StaleElementReferenceException) as e:
                last_exc = e
                fresh = getter_callable()
                driver.execute_script("arguments[0].value = arguments[1];", fresh, text)
                return
        except Exception as e:
            last_exc = e
            attach_screenshot(driver, f"safe_send_keys_retry_{{label}}_{{attempt+1}}")
            time.sleep(0.5)
    attach_screenshot(driver, f"safe_send_keys_failed_{{label}}")
    raise RuntimeError(f"safe_send_keys [{{label}}] failed: {{last_exc}}")

HELPER 5 — wait_for_sidebar_item (INCLUDE ONLY if any test clicks a sidebar/drawer/menu item):
def wait_for_sidebar_item(driver, getter_callable, timeout: int = 5):
    # CSS-animated sidebars report element_to_be_clickable=True mid-animation.
    # Polls getBoundingClientRect().width > 0 to confirm animation has completed.
    deadline = time.time() + timeout
    last_exc = None
    while time.time() < deadline:
        try:
            elem = getter_callable()
            width = driver.execute_script(
                "return arguments[0].getBoundingClientRect().width;", elem
            )
            if elem.is_displayed() and elem.is_enabled() and width > 0:
                return elem
        except Exception as e:
            last_exc = e
        time.sleep(0.2)
    raise RuntimeError(f"wait_for_sidebar_item timed out: {{last_exc}}")
